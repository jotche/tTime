# tTime with a Slint user interface (C++). On Windows it needs MSVC and Slint's prebuilt
# C++ package; see README.md:
#
#   cmake -S prototype/slint -B build/slint-prototype -G Ninja -DCMAKE_BUILD_TYPE=Release ^
#         -DCMAKE_PREFIX_PATH=%LOCALAPPDATA%/Slint/cpp-1.18.1
#   cmake --build build/slint-prototype
#   ctest --test-dir build/slint-prototype
cmake_minimum_required(VERSION 3.21)
project(ttime_slint VERSION 1.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Link the C++ runtime statically: ttime-slint.exe then needs no Visual C++
# Redistributable (slint_cpp.dll does not use it either), only Windows' own DLLs.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

find_package(Slint 1.18 REQUIRED)

set(ttime_resources ${CMAKE_CURRENT_SOURCE_DIR}/../../src/Resources)

# The alarm sound, compiled in as a byte array (played from memory).
set(sound_file ${ttime_resources}/sounds/submarine1.wav)
set(sound_header ${CMAKE_CURRENT_BINARY_DIR}/generated/alarm-sound-data.h)
if(${sound_file} IS_NEWER_THAN ${sound_header})
    file(READ ${sound_file} sound_hex HEX)
    string(LENGTH "${sound_hex}" sound_hex_length)
    math(EXPR sound_size "${sound_hex_length} / 2")
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," sound_bytes "${sound_hex}")
    string(REGEX REPLACE "(0x..,0x..,0x..,0x..,0x..,0x..,0x..,0x..,0x..,0x..,0x..,0x..,0x..,0x..,0x..,0x..,)" "\\1\n"
           sound_bytes "${sound_bytes}")
    file(WRITE ${sound_header}
         "// Generated by CMake from submarine1.wav.\n#pragma once\n#include <cstddef>\n#include <cstdint>\n"
         "alignas(4) inline const std::uint8_t ttime_alarm_sound[] = {\n${sound_bytes}\n};\n"
         "inline constexpr std::size_t ttime_alarm_sound_size = ${sound_size};\n")
endif()

# Everything but main(), shared by the program and the tests.
add_library(ttime_slint_core STATIC
    src/alarm-spec.cpp
    src/alarm.cpp
    src/bell-animation.cpp
    src/config.cpp
    src/platform.cpp
    src/settings.cpp
    src/sound.cpp
    src/theme.cpp
    src/tree-sway.cpp
    src/window-placement.cpp
    src/ui-support.cpp
    src/countdown-window.cpp
    src/time-up-window.cpp
    src/new-alarm-window.cpp
    src/alarm-manager.cpp)
slint_target_sources(ttime_slint_core ui/app.slint NAMESPACE ui)
target_include_directories(ttime_slint_core PUBLIC src ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(ttime_slint_core PUBLIC Slint::Slint)

if(MSVC)
    target_compile_options(ttime_slint_core PUBLIC /utf-8)
    # Strict warnings for our code only, not for the code Slint generates.
    get_target_property(core_sources ttime_slint_core SOURCES)
    list(FILTER core_sources INCLUDE REGEX "^src/")
    set_source_files_properties(${core_sources} main.cpp tests/logic-tests.cpp PROPERTIES COMPILE_OPTIONS "/W4;/wd4996")
endif()

if(WIN32)
    # winmm (PlaySoundW) is delay-loaded: it is only mapped when an alarm rings.
    # shcore: GetDpiForMonitor.
    target_link_libraries(ttime_slint_core PUBLIC winmm delayimp shcore)
    target_link_options(ttime_slint_core INTERFACE /DELAYLOAD:winmm.dll)
endif()

configure_file(ttime-slint.rc.in ${CMAKE_CURRENT_BINARY_DIR}/generated/ttime-slint.rc @ONLY)
add_executable(ttime-slint WIN32 main.cpp $<$<PLATFORM_ID:Windows>:${CMAKE_CURRENT_BINARY_DIR}/generated/ttime-slint.rc>)
target_link_libraries(ttime-slint PRIVATE ttime_slint_core)
target_compile_definitions(ttime-slint PRIVATE TTIME_VERSION="${PROJECT_VERSION}")
if(MSVC)
    # Keep main() as the entry point of the GUI-subsystem exe.
    target_link_options(ttime-slint PRIVATE /ENTRY:mainCRTStartup)
endif()

if(WIN32)
    # slint_cpp.dll and the default settings next to the exe.
    add_custom_command(TARGET ttime-slint POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_RUNTIME_DLLS:ttime-slint> $<TARGET_FILE_DIR:ttime-slint>
        COMMAND_EXPAND_LISTS)
endif()
add_custom_command(TARGET ttime-slint POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different ${CMAKE_CURRENT_SOURCE_DIR}/../../src/tTime.ini
            $<TARGET_FILE_DIR:ttime-slint>/tTime.ini)

# cmake --install: ttime-slint.exe, slint_cpp.dll, the default tTime.ini and the licences.
install(TARGETS ttime-slint RUNTIME DESTINATION .)
install(FILES $<TARGET_RUNTIME_DLLS:ttime-slint> ../../src/tTime.ini README.md DESTINATION .)
install(FILES ../../LICENSE ../../licenses/GPL-3.0.txt DESTINATION licenses)
file(GLOB font_licenses ${ttime_resources}/fonts/*-OFL.txt)
install(FILES ${font_licenses} DESTINATION licenses/fonts)
# Slint: GPLv3 / royalty-free / commercial (LICENSE.md), and its third-party notices.
get_filename_component(slint_licenses "${Slint_DIR}/../../../licenses" ABSOLUTE)
install(FILES ${slint_licenses}/LICENSE.md ${slint_licenses}/THIRDPARTY.md DESTINATION licenses/slint)

# cmake --build ... --target package: tTime-slint-<version>-windows-x64.zip
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_NAME tTime-slint)
set(CPACK_PACKAGE_FILE_NAME "tTime-slint-${PROJECT_VERSION}-windows-x64")
set(CPACK_PACKAGE_DIRECTORY ${CMAKE_BINARY_DIR}/packages)
include(CPack)

enable_testing()
add_executable(ttime-slint-tests tests/logic-tests.cpp)
target_link_libraries(ttime-slint-tests PRIVATE ttime_slint_core)
target_compile_definitions(ttime-slint-tests PRIVATE TTIME_RESOURCES="${ttime_resources}")
add_test(NAME logic COMMAND ttime-slint-tests)
